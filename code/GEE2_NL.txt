// Load the nightlights dataset
var nightlights = ee.Image('projects/[anonymized]/assets/nightlights_harmonized');
var newBands = ee.Image('projects/[anonymized]/assets/combinedNL_2022_2024');
var newBandNames = ['b31', 'b32', 'b33'];  // Replace with meaningful names if you want
var newBandsRenamed = newBands.rename(newBandNames);

var nightlights = nightlights.addBands(newBandsRenamed);


// Load the polygons (e.g., tang2023)
var tang2023 = ee.FeatureCollection("projects/[anonymized]/assets/tang2023");

var buffer100km = tang2023.map(function(feature) {
  var centroid = feature.geometry().centroid();
  var buffer = centroid.buffer(100000);  // 100 km
  return feature.setGeometry(buffer);
});

// SINGLE reduceRegions call
var result = nightlights.reduceRegions({
  collection: buffer100km,
  // collection: donuts,
  // collection: tang2023, // this one: for MeanNightLightsPolygon
  reducer: ee.Reducer.mean(),
  scale: 1000
});

// Optionally, export the results as a table if needed
