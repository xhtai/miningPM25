// Hansen GFC dataset
var gfc = ee.Image("UMD/hansen/global_forest_change_2024_v1_12");

// Select loss year band
var lossyear = gfc.select('lossyear');

// Mining polygons
var mines = ee.FeatureCollection("projects/[anonymized]/assets/tang2023");

// Pixel area (for counting pixels)
var pixelArea = ee.Image.constant(1).rename('total_pixels');

var years = ee.List.sequence(1, 24);  // 1 = 2001, 24 = 2024

var lossBands = ee.ImageCollection.fromImages(
  years.map(function(y){
    y = ee.Number(y);
    return lossyear.eq(y)
      .rename(ee.String('loss_').cat(y.add(2000).int().format()));
  })
).toBands();

var metricsImage = pixelArea.addBands(lossBands);

var polygonStats = metricsImage.reduceRegions({
  collection: mines,
  reducer: ee.Reducer.sum(),
  scale: 30
  // ,
  // tileScale: 4  // prevents memory errors for large polygons
});

Export.table.toDrive({
  collection: polygonStats,
  description: 'YearlyLossByPolygon',
  fileFormat: 'CSV'
});
